  try {

export function useSupabaseData() {
    const { state, setAppState: setState, loading, setLoading, session } = useAppStore();

    const fetchData = useCallback(async (isSilent = false) => {
        if (!session?.user?.id) return;
        if (!isSilent) setLoading(true);
        try {
            const results = await Promise.all([
                supabase.from('perfiles').select('*, centros!perfiles_centro_id_fkey(*)'),
                supabase.from('cursos').select('*'),
                supabase.from('estudiantes').select('*'),
                supabase.from('actividades').select('*'),
                supabase.from('calificaciones').select('*'),
                supabase.from('recuperaciones').select('*'),
                // ... 19 tablas adicionales
            ]);

            // Detección y reporte de errores por tabla
            results.forEach((result, idx) => {
                if (result.error) {
                    console.error(`[fetchData] Error en tabla "${tableNames[idx]}":`, result.error);
                }
            });

            // Protección: si una consulta falla, conservar estado previo
            const calificaciones = results[4].error ? null : results[4].data;
            const recuperaciones = results[5].error ? null : results[5].data;

            setState(prev => ({
                ...prev,
                // Mapeo de cada tabla con transformación snake_case → camelCase
                calificaciones: calificaciones === null ? prev.calificaciones
                    : calificaciones.map((c) => ({ /* ... mapeo ... */ })),
                recuperaciones: recuperaciones === null ? prev.recuperaciones
                    : recuperaciones.map((r) => ({ /* ... mapeo ... */ })),
            }));
        } finally { if (!isSilent) setLoading(false); }
    }, [session]);

    // Suscripciones Realtime a cambios en la base de datos
    useEffect(() => {
        if (session) {
            fetchData();
            const channel = supabase.channel(`db-changes-${session.user.id}`)
                .on('postgres_changes', { event: '*', schema: 'public', table: 'calificaciones' }, () => fetchData(true))
                .on('postgres_changes', { event: '*', schema: 'public', table: 'recuperaciones' }, () => fetchData(true))
                .on('postgres_changes', { event: '*', schema: 'public', table: 'cursos' }, () => fetchData(true))
                .subscribe();
            return () => supabase.removeChannel(channel);
        }
    }, [session, fetchData]);
}

Descripción: Lógica central de negocio para persistir calificaciones y actas de recuperación de competencias. Implementa upsert con resolución de conflictos por clave compuesta (estudiante + actividad) y actualización optimista del estado.

const saveCalificaciones = useCallback(async (
    califs: CalificacionActividad[],
    recs: RecuperacionBC[],
    cursoIdOverride?: number | null,
) => {
    const cursoId = cursoIdOverride ?? selectedCursoId;
    if (cursoId === null || !session?.user?.id) return;

    const dbCalifs = califs.map(c => ({
        user_id: session.user.id,
        curso_id: cursoId,
        estudiante_id: c.estudianteId,
        actividad_id: c.actividadId,
        asignatura: c.asignatura,
        periodo: c.periodo,
        competencias: c.competencias,
        descriptores: c.descriptores || [],
        puntaje: c.puntaje,
        recuperacion: c.recuperacion,
        shared_course_id: c.sharedCourseId || String(cursoId)
    }));

    const dbRecs = recs.map(r => ({
        user_id: session.user.id,
        estudiante_id: r.estudianteId,
        curso_id: r.cursoId,
        bc: r.bc,
        puntaje: r.puntaje,
        periodo: r.periodo,
        shared_course_id: r.sharedCourseId || String(r.cursoId),
        asignatura: r.asignatura || ''
    }));

    const promises = [];
    if (dbCalifs.length > 0)
        promises.push(supabase.from('calificaciones').upsert(dbCalifs,
            { onConflict: 'estudiante_id,actividad_id' }));
    if (dbRecs.length > 0)
        promises.push(supabase.from('recuperaciones').upsert(dbRecs,
            { onConflict: 'estudiante_id,curso_id,bc,periodo,asignatura' }));

    if (promises.length > 0) {
        await Promise.all(promises);
        // Actualización optimista del estado global
        setState(s => {
            const nextCalifs = [...s.calificaciones];
            califs.forEach(newC => {
                const idx = nextCalifs.findIndex(oc =>
                    oc.estudianteId === newC.estudianteId && oc.actividadId === newC.actividadId);
                if (idx !== -1) nextCalifs[idx] = newC; else nextCalifs.push(newC);
            });
            const nextRecs = [...s.recuperaciones];
            recs.forEach(nr => {
                const idx = nextRecs.findIndex(or =>
                    or.estudianteId === nr.estudianteId && or.bc === nr.bc
                    && or.periodo === nr.periodo && or.asignatura === nr.asignatura);
                if (idx !== -1) nextRecs[idx] = nr; else nextRecs.push(nr);
            });
            return { ...s, calificaciones: nextCalifs, recuperaciones: nextRecs };
        });
    }
}, [session, selectedCursoId, state, setState]);

Descripción: Esquema de políticas de seguridad a nivel de fila (Row Level Security) que garantiza aislamiento de datos por usuario. Incluye la función personalizada is_course_teacher() que verifica pertenencia a un curso como propietario o co-docente.

-- Función auxiliar para configurar RLS estándar en cada tabla
CREATE OR REPLACE FUNCTION setup_user_rls(table_name TEXT) RETURNS VOID AS $$
BEGIN
    EXECUTE format('ALTER TABLE public.%I ENABLE ROW LEVEL SECURITY', table_name);
    EXECUTE format('CREATE POLICY %I_select ON public.%I FOR SELECT
        TO authenticated USING (auth.uid() = user_id)', table_name, table_name);
    EXECUTE format('CREATE POLICY %I_insert ON public.%I FOR INSERT
        TO authenticated WITH CHECK (auth.uid() = user_id)', table_name, table_name);
    EXECUTE format('CREATE POLICY %I_update ON public.%I FOR UPDATE
        TO authenticated USING (auth.uid() = user_id)
        WITH CHECK (auth.uid() = user_id)', table_name, table_name);
END;
$$ LANGUAGE plpgsql;

-- Función de verificación de pertenencia a curso (SECURITY DEFINER)
CREATE OR REPLACE FUNCTION public.is_course_teacher(p_curso_id bigint)
RETURNS boolean AS $$
DECLARE
    current_user_id uuid;
BEGIN
    current_user_id := auth.uid();
    RETURN EXISTS (
        SELECT 1 FROM public.cursos
        WHERE id = p_curso_id AND user_id = current_user_id
    ) OR EXISTS (
        SELECT 1 FROM public.curso_docentes
        WHERE curso_id = p_curso_id AND docente_id = current_user_id
    );
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Tabla de tokens de recuperación de contraseña
CREATE TABLE IF NOT EXISTS public.recuperacion_tokens (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE NOT NULL,
    token TEXT UNIQUE NOT NULL,
    created_at TIMESTAMPTZ DEFAULT now() NOT NULL,
    expires_at TIMESTAMPTZ NOT NULL,
    estado TEXT DEFAULT 'activo' NOT NULL
        CHECK (estado IN ('activo', 'usado', 'revocado'))
);

-- Función para revocar tokens activos de un usuario (prevención de fugas)
CREATE OR REPLACE FUNCTION public.revocar_tokens_activos(p_user_id uuid)
RETURNS void AS $$
BEGIN
    UPDATE public.recuperacion_tokens
    SET estado = 'revocado'
    WHERE user_id = p_user_id AND estado = 'activo';
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
